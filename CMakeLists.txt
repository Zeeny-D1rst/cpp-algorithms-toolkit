cmake_minimum_required(VERSION 3.16)
project(cpp_algorithms_toolkit VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(algorithms_demo src/main.cpp)
target_include_directories(algorithms_demo PRIVATE include)
add_executable(algorithms_tests tests/algorithms_tests.cpp)
target_include_directories(algorithms_tests PRIVATE include)
enable_testing()
add_test(NAME algorithms_tests COMMAND algorithms_tests)
